func TestPass20Inventory(t *testing.T) {
	f := newPass19Fixture(t)
	f.capture(t)
	for position, step := range f.session.plan.steps {
		value := f.session.plan.values[step.Outputs[0]]
		t.Logf("node=%02d op=%s inputs=%v output=%s dtype=%v shape=%v declarations=%v retained=%t immutable_result=%t", position, step.Operator, step.Inputs, value.Name, value.Info.DataType, value.Info.Dimensions, value.Declarations, f.session.destinations.values[position] != nil, f.session.plan.calls[position].retained != nil)
	}
	for _, value := range f.session.plan.values {
		if value.Producer < 0 {
			t.Logf("binding=%s input=%t default=%t info=%+v declarations=%v", value.Name, value.Input, value.Default != nil, value.Info, value.Declarations)
		}
	}
}

// BenchmarkPass20Metadata measures the nine actually retained destinations.
// Every scope traverses the same slots; the table-only scope is its loop floor.
func BenchmarkPass20Metadata(b *testing.B) {
	f := newPass19Fixture(b)
	f.capture(b)
	var (
		values   []*tensor.Tensor
		shapes   [][]int64
		expected [][]int64
	)

	for _, value := range f.session.destinations.values {
		if value != nil {
			values = append(values, value)
			shapes = append(shapes, value.Shape())
			expected = append(expected, value.Shape())
		}
	}
	for _, mode := range []string{"Table", "Validate", "DataType", "ShapeLocal", "ShapeOwned", "Rank", "Len", "ByteSize", "CompareExact", "CopyLocal"} {
		b.Run(mode, func(part *testing.B) {
			var (
				value    *tensor.Tensor
				dataType DataType
				shape    []int64
				count    int64
				equal    bool
				buffer   [32]int64
				err      error
			)

			part.ReportAllocs()
			for part.Loop() {
				for index, candidate := range values {
					value = candidate
					switch mode {
					case "Validate":
						err = value.Validate()
					case "DataType":
						dataType = value.DataType()
					case "ShapeLocal":
						equal = slices.Equal(value.Shape(), shapes[index])
					case "ShapeOwned":
						shape = value.Shape()
					case "Rank":
						count = int64(value.Rank())
					case "Len":
						count = value.Len()
					case "ByteSize":
						count = value.ByteSize()
					case "CompareExact":
						equal = slices.Equal(shapes[index], expected[index])
					case "CopyLocal":
						copy(buffer[:], shapes[index])
					}
					if err != nil {
						part.Fatal(err)
					}
				}
			}
			runtime.KeepAlive(value)
			runtime.KeepAlive(dataType)
			runtime.KeepAlive(shape)
			runtime.KeepAlive(count)
			runtime.KeepAlive(equal)
			runtime.KeepAlive(buffer)
		})
	}
}
