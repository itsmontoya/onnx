func newPass17Fixture(tb testing.TB) (out *pass17Fixture) {
	var (
		f     pass17Fixture
		input *Tensor
		err   error
	)

	f.ctx = context.Background()
	if f.model, err = LoadFile(f.ctx, ".cache/compare-runtime/mnist-ir8-opset13.onnx", Options{}); err != nil {
		tb.Fatal(err)
	}

	tb.Cleanup(func() { _ = f.model.Close() })
	if f.session, err = f.model.NewSession(); err != nil {
		tb.Fatal(err)
	}

	tb.Cleanup(func() { _ = f.session.Close() })
	values := make([]float32, 784)
	for index := range values {
		values[index] = float32(index%256) / 255
	}
	if input, err = NewFloat32([]int64{1, 1, 28, 28}, values); err != nil {
		tb.Fatal(err)
	}

	f.inputs = map[string]*Tensor{"Input3": input}
	return &f
}

// pass17Fixture retains the ordinary destination reuse topology for diagnostics.
type pass17Fixture struct {
	ctx         context.Context
	model       *Model
	session     *Session
	inputs      map[string]*Tensor
	calls       []func() error
	callIndices []int
}

func (f *pass17Fixture) capture(tb testing.TB) (run *evaluation) {
	var err error
	if run, err = newEvaluation(f.ctx, f.session.plan); err != nil {
		tb.Fatal(err)
	}

	if err = run.bind(f.inputs); err != nil {
		tb.Fatal(err)
	}

	pass17UseDestinations = false
	pass17Destinations = nil
	pass17Arguments = nil
	for position, step := range f.session.plan.steps {
		arguments := make([]*tensor.Tensor, len(step.Inputs))
		for slot, index := range step.Inputs {
			if index >= 0 {
				arguments[slot] = run.storage.values[index]
			}
		}
		pass17Arguments = append(pass17Arguments, arguments)
		if err = run.execute(step); err != nil {
			tb.Fatal(err)
		}

		pass17Destinations = append(pass17Destinations, run.storage.values[step.Outputs[0]])
		destination := run.storage.values[step.Outputs[0]]
		prepared := f.session.plan.calls[step.Index]
		if prepared.retained == nil {
			f.callIndices = append(f.callIndices, position)
			switch {
			case prepared.cnn != nil:
				call := *prepared.cnn
				call.inputs = arguments
				f.calls = append(f.calls, func() (callErr error) {
					_, callErr = call.run(f.ctx, destination)
					return callErr
				})
			case prepared.numerical != nil:
				call := *prepared.numerical
				call.inputs = arguments
				f.calls = append(f.calls, func() (callErr error) {
					_, callErr = call.run(f.ctx, destination)
					return callErr
				})
			default:
				call := *prepared.shape
				call.inputs = arguments
				f.calls = append(f.calls, func() (callErr error) {
					_, callErr = call.run(f.ctx, destination)
					return callErr
				})
			}
		}
		run.storage.after(position)
	}
	run.storage.trim()
	return run
}

// BenchmarkPass17Prepared follows the current retained-constant behavior and
// the ordinary intra-Run destination alias topology, with all setup excluded.
func BenchmarkPass17Prepared(b *testing.B) {
	var (
		fixture  = newPass17Fixture(b)
		run      = fixture.capture(b)
		output   = run.storage.values[fixture.session.plan.outputIndices[0]]
		expected []float32
		actual   []float32
		err      error
	)

	if expected, err = output.Float32Values(); err != nil {
		b.Fatal(err)
	}

	b.Run("Traversal", func(part *testing.B) {
		part.ReportAllocs()
		for part.Loop() {
			for _, call := range fixture.calls {
				if err = call(); err != nil {
					part.Fatal(err)
				}
			}
		}
	})
	if actual, err = output.Float32Values(); err != nil {
		b.Fatal(err)
	}

	for index, value := range expected {
		if math.Float32bits(value) != math.Float32bits(actual[index]) {
			b.Fatal("prepared traversal changed output bits")
		}
	}
	for position, call := range fixture.calls {
		index := fixture.callIndices[position]
		if index != 2 && index != 6 && index != 8 {
			continue
		}

		b.Run(fmt.Sprintf("Node%02d", index), func(part *testing.B) {
			part.ReportAllocs()
			for part.Loop() {
				if err = call(); err != nil {
					part.Fatal(err)
				}
			}
		})
	}
}

func BenchmarkPass17Run(b *testing.B) {
	var (
		fixture = newPass17Fixture(b)
		err     error
	)

	fixture.capture(b)
	pass17UseDestinations = true
	b.ReportAllocs()
	b.ResetTimer()
	for b.Loop() {
		pass17Next = 0
		if _, err = fixture.session.Run(fixture.ctx, fixture.inputs, RunOptions{}); err != nil {
			b.Fatal(err)
		}
	}
}

func BenchmarkPass17Parts(b *testing.B) {
	var (
		fixture = newPass17Fixture(b)
		run     = fixture.capture(b)
		source  = run.storage.values[fixture.session.plan.outputIndices[0]]
		err     error
	)

	for _, mode := range []string{"Publish", "Clone", "BackendClone", "PayloadCopy", "Arguments", "CheckValues", "Destination", "Retirement", "ContextChecks"} {
		b.Run(mode, func(part *testing.B) {
			var (
				arguments []*tensor.Tensor
				backend   *tensor.Tensor
				outputs   []NamedTensor
				wrapped   *Tensor
				values    []float32
			)

			if values, err = source.Float32Values(); err != nil {
				part.Fatal(err)
			}

			destination := make([]float32, len(values))
			part.ReportAllocs()
			part.ResetTimer()
			for part.Loop() {
				switch mode {
				case "Publish":
					run.budget.live = 0
					outputs, err = run.publish(fixture.session.plan.outputIndices)
				case "Clone":
					run.budget.live = 0
					wrapped, err = run.cloneOutput(source)
				case "BackendClone":
					backend, err = source.Clone()
				case "PayloadCopy":
					copy(destination, values)
				case "Arguments":
					for _, step := range fixture.session.plan.steps {
						arguments = make([]*tensor.Tensor, len(step.Inputs))
						for slot, index := range step.Inputs {
							if err = fixture.ctx.Err(); err != nil {
								part.Fatal(err)
							}

							if index >= 0 {
								arguments[slot] = run.storage.values[index]
							}
						}
					}
				case "CheckValues":
					for index, step := range fixture.session.plan.steps {
						err = run.checkValue(fixture.session.plan.values[step.Outputs[0]], pass17Destinations[index], ErrInvalidModel)
					}
				case "Destination":
					for index := range fixture.session.plan.steps {
						run.storage.free = pass17Destinations[index]
						run.budget.live = 0
						backend, err = run.storage.destination(pass17Destinations[index].DataType(), pass17Destinations[index].Shape(), pass17Destinations[index].ByteSize(), 0)
					}
				case "Retirement":
					for position := range fixture.session.plan.steps {
						run.storage.after(position)
					}
				case "ContextChecks":
					for range 100 {
						err = fixture.ctx.Err()
					}
				}
				if err != nil {
					part.Fatal(err)
				}
			}
			part.StopTimer()
			runtime.KeepAlive(arguments)
			runtime.KeepAlive(backend)
			runtime.KeepAlive(outputs)
			runtime.KeepAlive(wrapped)
			runtime.KeepAlive(destination)
		})
	}
}

func TestPass17Ledger(t *testing.T) {
	var (
		fixture       = newPass17Fixture(t)
		before, after runtime.MemStats
		run           *evaluation
		err           error
	)

	// Collect ordinary counters with no allocation-profile sampling or logging
	// inside each measured scope. Tiny identities can share allocator blocks.
	const count = 1000
	bytes := make([]uint64, len(fixture.session.plan.steps)+3)
	objects := make([]uint64, len(bytes))
	for range count {
		runtime.ReadMemStats(&before)
		run, err = newEvaluation(fixture.ctx, fixture.session.plan)
		runtime.ReadMemStats(&after)
		bytes[0] += after.TotalAlloc - before.TotalAlloc
		objects[0] += after.Mallocs - before.Mallocs
		if err != nil {
			t.Fatal(err)
		}

		runtime.ReadMemStats(&before)
		err = run.bind(fixture.inputs)
		runtime.ReadMemStats(&after)
		bytes[1] += after.TotalAlloc - before.TotalAlloc
		objects[1] += after.Mallocs - before.Mallocs
		if err != nil {
			t.Fatal(err)
		}

		for position, step := range fixture.session.plan.steps {
			runtime.ReadMemStats(&before)
			err = run.execute(step)
			runtime.ReadMemStats(&after)
			bytes[position+2] += after.TotalAlloc - before.TotalAlloc
			objects[position+2] += after.Mallocs - before.Mallocs
			if err != nil {
				t.Fatal(err)
			}

			run.storage.after(position)
		}
		run.storage.trim()
		runtime.ReadMemStats(&before)
		_, err = run.publish(fixture.session.plan.outputIndices)
		runtime.ReadMemStats(&after)
		bytes[len(bytes)-1] += after.TotalAlloc - before.TotalAlloc
		objects[len(bytes)-1] += after.Mallocs - before.Mallocs
		if err != nil {
			t.Fatal(err)
		}
	}
	for index := range bytes {
		t.Logf("scope=%d bytes=%.4f objects=%.4f", index, float64(bytes[index])/count, float64(objects[index])/count)
	}
}

func TestPass17Repeat(t *testing.T) {
	var (
		fixture  = newPass17Fixture(t)
		run      = fixture.capture(t)
		index    = fixture.session.plan.outputIndices[0]
		expected []float32
		actual   []float32
		err      error
	)

	if expected, err = run.storage.values[index].Float32Values(); err != nil {
		t.Fatal(err)
	}

	pass17UseDestinations = true
	for range 3 {
		pass17Next = 0
		if run, err = newEvaluation(fixture.ctx, fixture.session.plan); err != nil {
			t.Fatal(err)
		}

		if _, err = run.run(fixture.inputs, RunOptions{}); err != nil {
			t.Fatal(err)
		}

		if actual, err = run.storage.values[index].Float32Values(); err != nil {
			t.Fatal(err)
		}

		for position, value := range expected {
			if math.Float32bits(value) != math.Float32bits(actual[position]) {
				t.Fatal("diagnostic differs from ordinary traversal")
			}
		}
	}
}

	pass17Next            int
	pass17UseDestinations bool
	pass17Arguments       [][]*tensor.Tensor
)
