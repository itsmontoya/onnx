package tensor

import "testing"

// BenchmarkPass17Pool separates current public validation from the existing
// Pass 9 small-window kernel. It deliberately does not propose new arithmetic.
func BenchmarkPass17Pool(b *testing.B) {
	for _, size := range []int64{2, 3} {
		b.Run(map[int64]string{2: "Pool03", 3: "Pool07"}[size], func(pool *testing.B) {
			var (
				shape       = []int64{1, 8, 28, 28}
				config      = Pool2DConfig{Kernel: [2]int64{size, size}, Strides: [2]int64{size, size}, Dilations: [2]int64{1, 1}}
				outputShape []int64
				err         error
			)

			if size == 3 {
				shape = []int64{1, 16, 14, 14}
			}

			if outputShape, err = Pool2DShape(shape, config); err != nil {
				pool.Fatal(err)
			}

			input := pass16Tensor(pool, shape, 3)
			output := pass16Tensor(pool, outputShape, 0)
			for _, mode := range []string{"Complete", "Validation", "Shape", "Kernel"} {
				pool.Run(mode, func(part *testing.B) {
					part.ReportAllocs()
					for part.Loop() {
						switch mode {
						case "Complete":
							err = MaxPool2DInto(input, output, config)
						case "Validation":
							err = validatePoolDestination(input, output, config)
						case "Shape":
							outputShape, err = Pool2DShape(shape, config)
						case "Kernel":
							if !maxPool2DSmallWindowsInto(input, output, config) {
								part.Fatal("kernel not selected")
							}
						}
						if err != nil {
							part.Fatal(err)
						}
					}
				})
			}
		})
	}
}

func BenchmarkPass17Tensor(b *testing.B) {
	for _, mode := range []string{"NewActivation", "ClearActivation", "CopyActivation", "CopyReshape", "NewOutput", "OutputShape", "ValidateOutput"} {
		b.Run(mode, func(part *testing.B) {
			var (
				input  = pass16Tensor(part, []int64{1, 8, 28, 28}, 3)
				output = pass16Tensor(part, []int64{1, 8, 28, 28}, 0)
				result *Tensor
				shape  []int64
				err    error
			)

			part.ReportAllocs()
			for part.Loop() {
				switch mode {
				case "NewActivation":
					result, err = New(Float32, input.shape)
				case "ClearActivation":
					clear(output.floats)
				case "CopyActivation":
					copy(output.floats, input.floats)
				case "CopyReshape":
					copy(output.floats[:256], input.floats[:256])
				case "NewOutput":
					result, err = New(Float32, []int64{1, 10})
				case "OutputShape":
					shape = input.Shape()
				case "ValidateOutput":
					err = input.Validate()
				}
				if err != nil {
					part.Fatal(err)
				}
			}
			part.StopTimer()
			if mode == "OutputShape" && len(shape) != 4 || mode == "NewOutput" && result.Len() != 10 || mode == "CopyReshape" && output.floats[0] != input.floats[0] {
				part.Fatal("diagnostic output mismatch")
			}
		})
	}
}

func TestPass17PoolKernel(t *testing.T) {
	for _, size := range []int64{2, 3} {
		var (
			input  = pass16Tensor(t, []int64{1, 8, 28, 28}, 3)
			config = Pool2DConfig{Kernel: [2]int64{size, size}, Strides: [2]int64{size, size}, Dilations: [2]int64{1, 1}}
			shape  []int64
			err    error
		)

		if shape, err = Pool2DShape(input.shape, config); err != nil {
			t.Fatal(err)
		}

		complete := pass16Tensor(t, shape, 0)
		kernel := pass16Tensor(t, shape, 0)
		if err = MaxPool2DInto(input, complete, config); err != nil {
			t.Fatal(err)
		}

		if !maxPool2DSmallWindowsInto(input, kernel, config) {
			t.Fatal("kernel not selected")
		}

		pass16EqualBits(t, complete.floats, kernel.floats)
	}
}
