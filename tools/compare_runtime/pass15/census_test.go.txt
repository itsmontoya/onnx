package onnx
import ("context";"fmt";"os";"testing")
func TestPass15TensorCensus(t *testing.T) {
 ctx:=context.Background()
 m,err:=LoadFile(ctx,".cache/compare-runtime/mnist-ir8-opset13.onnx",Options{})
 if err!=nil {t.Fatal(err)}
 defer m.Close()
 s,err:=m.NewSession()
 if err!=nil {t.Fatal(err)}
 defer s.Close()
 input,err:=NewFloat32([]int64{1,1,28,28},make([]float32,784))
 if err!=nil {t.Fatal(err)}
 fmt.Fprintln(os.Stderr,"BEGIN_RUN")
 _,err=s.Run(ctx,map[string]*Tensor{"Input3":input},RunOptions{})
 fmt.Fprintln(os.Stderr,"END_RUN")
 if err!=nil {t.Fatal(err)}
}
