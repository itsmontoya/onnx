package tensor

import "testing"

var pass15Shape []int64

// BenchmarkPass15 isolates current validation/planning from the unchanged span loop.
func BenchmarkPass15(b *testing.B) {
	var tests = []struct {
		name        string
		left, right []int64
	}{
		{name: "Add1", left: []int64{1, 8, 28, 28}, right: []int64{8, 1, 1}},
		{name: "Add2", left: []int64{1, 16, 14, 14}, right: []int64{16, 1, 1}},
		{name: "Add3", left: []int64{1, 10}, right: []int64{1, 10}},
	}

	for _, tt := range tests {
		b.Run(tt.name, func(call *testing.B) {
			var (
				left, right, output *Tensor
				shape, ls, rs       []int64
				err                 error
			)

			if left, err = New(Float32, tt.left); err != nil {
				call.Fatal(err)
			}

			if right, err = New(Float32, tt.right); err != nil {
				call.Fatal(err)
			}

			for index := range left.floats {
				left.floats[index] = float32(index%29-14) * 0.125
			}
			for index := range right.floats {
				right.floats[index] = float32(index%13-6) * 0.25
			}
			if shape, err = BroadcastShape(tt.left, tt.right); err != nil {
				call.Fatal(err)
			}

			if output, err = New(Float32, shape); err != nil {
				call.Fatal(err)
			}

			if ls, err = broadcastStrides(tt.left, shape); err != nil {
				call.Fatal(err)
			}

			if rs, err = broadcastStrides(tt.right, shape); err != nil {
				call.Fatal(err)
			}

			span, axes := scalarBroadcastSpan(tt.left, tt.right, shape)
			for _, mode := range []string{"Complete", "Shape", "Destination", "Inputs", "Planning", "Arithmetic", "Prepared"} {
				call.Run(mode, func(part *testing.B) {
					part.ReportAllocs()
					for part.Loop() {
						switch mode {
						case "Complete":
							err = AddInto(left, right, output)
						case "Shape":
							pass15Shape, err = BroadcastShape(tt.left, tt.right)
						case "Destination":
							err = validateDestination(output, Float32, shape, left, right)
						case "Inputs":
							if err = validateTensorType("left", left, Float32); err == nil {
								err = validateTensorType("right", right, Float32)
							}
						case "Planning":
							pass15Shape, err = BroadcastShape(tt.left, tt.right)
							if err == nil && span > 1 {
								_, err = broadcastStrides(tt.left, shape)
								if err == nil {
									_, err = broadcastStrides(tt.right, shape)
								}
							}
						case "Prepared":
							if err = validateTensorType("left", left, Float32); err == nil {
								err = validateTensorType("right", right, Float32)
							}
							if err == nil && (!shapesEqual(left.shape, tt.left) || !shapesEqual(right.shape, tt.right)) {
								part.Fatal("prepared signature mismatch")
							}
							if err == nil {
								err = validateDestination(output, Float32, shape, left, right)
							}
							if err == nil {
								pass15Arithmetic(left, right, output, ls, rs, span, axes)
							}
						case "Arithmetic":
							pass15Arithmetic(left, right, output, ls, rs, span, axes)
						}
						if err != nil {
							part.Fatal(err)
						}
					}
				})
			}
		})
	}
}

// pass15Arithmetic is diagnostic only: equal shapes deliberately bypass matrix copies.
func pass15Arithmetic(left, right, destination *Tensor, leftStrides, rightStrides []int64, span, axes int) {
	if span == 1 {
		for index, value := range left.floats {
			destination.floats[index] = value + right.floats[index]
		}
		return
	}

	var (
		coordinates             = make([]int64, axes)
		leftOffset, rightOffset int64
	)

	for start := 0; start < len(destination.floats); start += span {
		var (
			output = destination.floats[start : start+span]
			values = left.floats[int(leftOffset) : int(leftOffset)+span]
			scalar = right.floats[int(rightOffset)]
		)

		for index, value := range values {
			output[index] = value + scalar
		}
		for axis := axes - 1; axis >= 0; axis-- {
			coordinates[axis]++
			if coordinates[axis] < destination.shape[axis] {
				leftOffset += leftStrides[axis]
				rightOffset += rightStrides[axis]
				break
			}

			coordinates[axis] = 0
			leftOffset -= (destination.shape[axis] - 1) * leftStrides[axis]
			rightOffset -= (destination.shape[axis] - 1) * rightStrides[axis]
		}
	}
}
