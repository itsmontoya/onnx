package tensor

import (
	"fmt"
	"github.com/itsmontoya/neuralnetwork/matrix"
	"slices"
)

func NewPass15AddPlan(left, right []int64) (out *Pass15AddPlan, err error) {
	var p Pass15AddPlan
	if p.shape, err = BroadcastShape(left, right); err != nil {
		return nil, err
	}
	p.left = slices.Clone(left)
	p.right = slices.Clone(right)
	p.equal = shapesEqual(left, right)
	if p.equal {
		return &p, nil
	}
	p.span, p.axes = scalarBroadcastSpan(left, right, p.shape)
	if p.span == 1 {
		p.span, p.axes = scalarBroadcastSpan(right, left, p.shape)
		p.scalarLeft = true
	}
	if p.span == 1 {
		return nil, fmt.Errorf("diagnostic: unhandled geometry")
	}
	if p.leftStrides, err = broadcastStrides(left, p.shape); err != nil {
		return nil, err
	}
	if p.rightStrides, err = broadcastStrides(right, p.shape); err != nil {
		return nil, err
	}
	return &p, nil
}

// Pass15AddPlan is diagnostic immutable metadata, with no retained tensors.
type Pass15AddPlan struct {
	left, right, shape, leftStrides, rightStrides []int64
	span, axes                                    int
	scalarLeft, equal                             bool
}

func (p *Pass15AddPlan) Into(left, right, destination *Tensor) (err error) {
	if err = validateTensorType("left", left, Float32); err != nil {
		return err
	}
	if err = validateTensorType("right", right, Float32); err != nil {
		return err
	}
	if !shapesEqual(left.shape, p.left) || !shapesEqual(right.shape, p.right) {
		return fmt.Errorf("diagnostic: signature mismatch")
	}
	if err = validateDestination(destination, Float32, p.shape, left, right); err != nil {
		return err
	}
	if len(destination.floats) == 0 {
		return nil
	}
	if p.equal {
		return matrixBinaryInto(left, right, destination, (*matrix.Matrix).AddInto)
	}
	var (
		span, axes                = p.span, p.axes
		scalarLeft                = p.scalarLeft
		leftStrides, rightStrides = p.leftStrides, p.rightStrides
		coordinates               = make([]int64, axes)
		leftOffset, rightOffset   int64
	)
	for start := 0; start < len(destination.floats); start += span {
		var output = destination.floats[start : start+span]
		if scalarLeft {
			var (
				scalar = left.floats[int(leftOffset)]
				values = right.floats[int(rightOffset) : int(rightOffset)+span]
			)
			for index, value := range values {
				output[index] = scalar + value
			}
		} else {
			var (
				values = left.floats[int(leftOffset) : int(leftOffset)+span]
				scalar = right.floats[int(rightOffset)]
			)
			for index, value := range values {
				output[index] = value + scalar
			}
		}
		// Only the prefix carries: the suffix was consumed as one contiguous span.
		for axis := axes - 1; axis >= 0; axis-- {
			coordinates[axis]++
			if coordinates[axis] < destination.shape[axis] {
				leftOffset += leftStrides[axis]
				rightOffset += rightStrides[axis]
				break
			}
			coordinates[axis] = 0
			leftOffset -= (destination.shape[axis] - 1) * leftStrides[axis]
			rightOffset -= (destination.shape[axis] - 1) * rightStrides[axis]
		}
	}
	return nil
}
