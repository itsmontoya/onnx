// BenchmarkPass18 measures session construction, the indicated number of calls,
// and deterministic Close; model loading and input construction are excluded.
func BenchmarkPass18(b *testing.B) {
	ctx := context.Background()
	model, err := LoadFile(ctx, ".cache/compare-runtime/mnist-ir8-opset13.onnx", Options{})
	if err != nil {
		b.Fatal(err)
	}

	defer model.Close()
	input, err := NewFloat32([]int64{1, 1, 28, 28}, make([]float32, 784))
	if err != nil {
		b.Fatal(err)
	}

	inputs := map[string]*Tensor{"Input3": input}
	for _, runs := range []int{1, 2, 10, 100} {
		b.Run(fmt.Sprint(runs), func(part *testing.B) {
			part.ReportAllocs()
			for part.Loop() {
				session, sessionErr := model.NewSession()
				if sessionErr != nil {
					part.Fatal(sessionErr)
				}

				for repeat := 0; repeat < runs; repeat++ {
					if _, sessionErr = session.Run(ctx, inputs, RunOptions{}); sessionErr != nil {
						part.Fatal(sessionErr)
					}
				}
				if sessionErr = session.Close(); sessionErr != nil {
					part.Fatal(sessionErr)
				}
			}
		})
	}
}

func BenchmarkPass18ColdAndSession(b *testing.B) {
	ctx := context.Background()
	model, err := LoadFile(ctx, ".cache/compare-runtime/mnist-ir8-opset13.onnx", Options{})
	if err != nil {
		b.Fatal(err)
	}

	defer model.Close()
	input, err := NewFloat32([]int64{1, 1, 28, 28}, make([]float32, 784))
	if err != nil {
		b.Fatal(err)
	}

	inputs := map[string]*Tensor{"Input3": input}
	b.Run("Session", func(part *testing.B) {
		part.ReportAllocs()
		for part.Loop() {
			session, sessionErr := model.NewSession()
			if sessionErr != nil {
				part.Fatal(sessionErr)
			}

			if sessionErr = session.Close(); sessionErr != nil {
				part.Fatal(sessionErr)
			}
		}
	})
	b.Run("ColdRun", func(part *testing.B) {
		part.ReportAllocs()
		for part.Loop() {
			part.StopTimer()
			session, sessionErr := model.NewSession()
			if sessionErr != nil {
				part.Fatal(sessionErr)
			}

			part.StartTimer()
			if _, sessionErr = session.Run(ctx, inputs, RunOptions{}); sessionErr != nil {
				part.Fatal(sessionErr)
			}

			part.StopTimer()
			if sessionErr = session.Close(); sessionErr != nil {
				part.Fatal(sessionErr)
			}

			part.StartTimer()
		}
	})
}

func BenchmarkPass18Workloads(b *testing.B) {
	ctx := context.Background()
	model, err := LoadFile(ctx, ".cache/compare-runtime/mnist-ir8-opset13.onnx", Options{})
	if err != nil {
		b.Fatal(err)
	}

	defer model.Close()
	input, err := NewFloat32([]int64{1, 1, 28, 28}, make([]float32, 784))
	if err != nil {
		b.Fatal(err)
	}

	inputs := map[string]*Tensor{"Input3": input}
	b.Run("AlternatingSessions", func(part *testing.B) {
		var sessions [2]*Session
		for index := range sessions {
			if sessions[index], err = model.NewSession(); err != nil {
				part.Fatal(err)
			}

			if _, err = sessions[index].Run(ctx, inputs, RunOptions{}); err != nil {
				part.Fatal(err)
			}
		}
		defer sessions[0].Close()
		defer sessions[1].Close()
		part.ReportAllocs()
		part.ResetTimer()
		next := 0
		for part.Loop() {
			if _, err = sessions[next%2].Run(ctx, inputs, RunOptions{}); err != nil {
				part.Fatal(err)
			}

			next++
		}
	})
	b.Run("ConcurrentSessions", func(part *testing.B) {
		part.ReportAllocs()
		part.RunParallel(func(worker *testing.PB) {
			session, sessionErr := model.NewSession()
			if sessionErr != nil {
				part.Fatal(sessionErr)
			}

			defer session.Close()
			if _, sessionErr = session.Run(ctx, inputs, RunOptions{}); sessionErr != nil {
				part.Fatal(sessionErr)
			}

			for worker.Next() {
				if _, sessionErr = session.Run(ctx, inputs, RunOptions{}); sessionErr != nil {
					part.Fatal(sessionErr)
				}
			}
		})
	})
	b.Run("BindingFailureRecovery", func(part *testing.B) {
		session, sessionErr := model.NewSession()
		if sessionErr != nil {
			part.Fatal(sessionErr)
		}

		defer session.Close()
		if _, sessionErr = session.Run(ctx, inputs, RunOptions{}); sessionErr != nil {
			part.Fatal(sessionErr)
		}

		part.ReportAllocs()
		part.ResetTimer()
		for part.Loop() {
			if _, sessionErr = session.Run(ctx, nil, RunOptions{}); sessionErr == nil {
				part.Fatal("missing input accepted")
			}

			if _, sessionErr = session.Run(ctx, inputs, RunOptions{}); sessionErr != nil {
				part.Fatal(sessionErr)
			}
		}
	})
	b.Run("CancelRecovery", func(part *testing.B) {
		session, sessionErr := model.NewSession()
		if sessionErr != nil {
			part.Fatal(sessionErr)
		}

		defer session.Close()
		if _, sessionErr = session.Run(ctx, inputs, RunOptions{}); sessionErr != nil {
			part.Fatal(sessionErr)
		}

		cancelled, cancel := context.WithCancel(ctx)
		cancel()
		part.ReportAllocs()
		part.ResetTimer()
		for part.Loop() {
			if _, sessionErr = session.Run(cancelled, inputs, RunOptions{}); sessionErr == nil {
				part.Fatal("cancel accepted")
			}

			if _, sessionErr = session.Run(ctx, inputs, RunOptions{}); sessionErr != nil {
				part.Fatal(sessionErr)
			}
		}
	})
}

func BenchmarkPass18DynamicShapes(b *testing.B) {
	data, err := os.ReadFile("testdata/profile/identity.onnx")
	if err != nil {
		b.Fatal(err)
	}

	var wire onnxpb.ModelProto
	if err = proto.Unmarshal(data, &wire); err != nil {
		b.Fatal(err)
	}

	wire.Graph.Node = []*onnxpb.NodeProto{
		{OpType: proto.String("Add"), Input: []string{"x", "x"}, Output: []string{"a"}},
		{OpType: proto.String("Relu"), Input: []string{"a"}, Output: []string{"y"}},
	}
	if data, err = proto.Marshal(&wire); err != nil {
		b.Fatal(err)
	}

	ctx := context.Background()
	model, err := LoadBytes(ctx, data, Options{})
	if err != nil {
		b.Fatal(err)
	}

	defer model.Close()
	session, err := model.NewSession()
	if err != nil {
		b.Fatal(err)
	}

	defer session.Close()
	var cases []map[string]*Tensor
	for _, batch := range []int64{1, 3, 0, 100, 2} {
		input, inputErr := NewFloat32([]int64{batch, 2}, make([]float32, batch*2))
		if inputErr != nil {
			b.Fatal(inputErr)
		}

		cases = append(cases, map[string]*Tensor{"x": input})
	}
	b.ReportAllocs()
	b.ResetTimer()
	next := 0
	for b.Loop() {
		if _, err = session.Run(ctx, cases[next%len(cases)], RunOptions{}); err != nil {
			b.Fatal(err)
		}

		next++
	}
}

func BenchmarkPass18Initialization(b *testing.B) {
	for _, mode := range []string{"New", "Clear", "Copy"} {
		b.Run(mode, func(part *testing.B) {
			values := make([]float32, 6272)
			destination := make([]float32, len(values))
			var result *tensor.Tensor
			part.ReportAllocs()
			part.ResetTimer()
			for part.Loop() {
				switch mode {
				case "New":
					var err error
					if result, err = tensor.New(Float32, []int64{1, 8, 28, 28}); err != nil {
						part.Fatal(err)
					}
				case "Clear":
					clear(destination)
				case "Copy":
					copy(destination, values)
				}
			}
			runtime.KeepAlive(result)
			runtime.KeepAlive(destination)
		})
	}
}
