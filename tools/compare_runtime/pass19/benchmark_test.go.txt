// BenchmarkPass19State times inference only; session preparation and the first
// invocation for SecondRun are excluded, including their allocation counters.
func BenchmarkPass19State(b *testing.B) {
	for _, state := range []string{"ColdRun", "SecondRun"} {
		b.Run(state, func(part *testing.B) {
			f := newPass19Fixture(part)
			if err := f.session.Close(); err != nil {
				part.Fatal(err)
			}

			part.ReportAllocs()
			part.ResetTimer()
			for part.Loop() {
				part.StopTimer()
				var err error
				if f.session, err = f.model.NewSession(); err != nil {
					part.Fatal(err)
				}

				if state == "SecondRun" {
					if _, err = f.session.Run(f.ctx, f.inputs, RunOptions{}); err != nil {
						part.Fatal(err)
					}
				}
				part.StartTimer()
				if _, err = f.session.Run(f.ctx, f.inputs, RunOptions{}); err != nil {
					part.Fatal(err)
				}

				part.StopTimer()
				if err = f.session.Close(); err != nil {
					part.Fatal(err)
				}

				part.StartTimer()
			}
		})
	}
}

func TestPass19Retention(t *testing.T) {
	var (
		f        = newPass19Fixture(t)
		previous []*tensor.Tensor
		lastLive int64
	)

	if f.session.destinations.bytes != 0 {
		t.Fatal("cold session already populated")
	}

	for repeat := range 1000 {
		run := f.evaluation(t)
		if _, err := run.run(f.inputs, RunOptions{}); err != nil {
			t.Fatal(err)
		}

		count := 0
		var bytes int64
		for slot, value := range f.session.destinations.values {
			if value == nil {
				continue
			}

			count++
			bytes += value.ByteSize()
			if repeat != 0 && previous[slot] != value {
				t.Fatal("retained slot replacement")
			}
		}
		if count != 9 || bytes != 83664 || bytes != f.session.destinations.bytes || run.budget.workspace != 0 {
			t.Fatalf("retention mismatch count=%d bytes=%d budget=%+v", count, bytes, run.budget)
		}

		if repeat == 0 {
			previous = append([]*tensor.Tensor(nil), f.session.destinations.values...)
			lastLive = run.budget.live
		}

		if run.budget.live != lastLive {
			t.Fatal("unbalanced accounting between invocations")
		}

		if repeat < 3 || repeat == 999 {
			t.Logf("run=%d slots=%d retained=%d live=%d workspace=%d intra_reuses=%d limit=%d resident=%d", repeat+1, count, bytes, run.budget.live, run.budget.workspace, run.storage.reuses, f.session.destinations.limit, f.session.plan.residentBytes)
		}
	}
}

func TestPass19Repeat(t *testing.T) {
	var (
		f        = newPass19Fixture(t)
		run      = f.capture(t)
		source   = run.storage.values[f.session.plan.outputIndices[0]]
		expected []float32
		actual   []float32
		err      error
	)

	if expected, err = source.Float32Values(); err != nil {
		t.Fatal(err)
	}

	for range 10 {
		for _, call := range f.calls {
			if err = call(); err != nil {
				t.Fatal(err)
			}
		}
		if actual, err = source.Float32Values(); err != nil {
			t.Fatal(err)
		}

		for index, value := range expected {
			if math.Float32bits(value) != math.Float32bits(actual[index]) {
				t.Fatal("prepared traversal changed output bits")
			}
		}
	}
	counter := pass19Context{Context: context.Background()}
	if _, err = f.session.Run(&counter, f.inputs, RunOptions{}); err != nil {
		t.Fatal(err)
	}

	t.Logf("context checks per Run=%d", counter.checks)
}

func BenchmarkPass19Parts(b *testing.B) {
	var (
		f      = newPass19Fixture(b)
		run    = f.capture(b)
		source = run.storage.values[f.session.plan.outputIndices[0]]
		shapes = make([][]int64, len(f.outputs))
		bytes  = make([]int64, len(f.outputs))
		err    error
	)

	for index, value := range f.outputs {
		shapes[index], bytes[index] = value.Shape(), value.ByteSize()
	}

	for _, mode := range []string{"Evaluation", "Bind", "Arguments", "CheckValues", "Publish", "Clone", "BackendClone", "PayloadCopy", "RetainedLookup", "DirectLookup", "Acquisition", "Node09Dispatch", "ContextChecks"} {
		b.Run(mode, func(part *testing.B) {
			var (
				arguments []*tensor.Tensor
				backend   *tensor.Tensor
				outputs   []NamedTensor
				wrapped   *Tensor
				created   *evaluation
				values    []float32
			)

			if values, err = source.Float32Values(); err != nil {
				part.Fatal(err)
			}

			destination := make([]float32, len(values))
			part.ReportAllocs()
			part.ResetTimer()
			for part.Loop() {
				run.budget.live = f.session.plan.residentBytes + f.session.destinations.bytes
				switch mode {
				case "Evaluation":
					created, err = newEvaluation(f.ctx, f.session.plan)
				case "Bind":
					err = run.bind(f.inputs)
				case "Arguments":
					for _, step := range f.session.plan.steps {
						arguments = make([]*tensor.Tensor, len(step.Inputs))
						for slot, index := range step.Inputs {
							if err = f.ctx.Err(); err != nil {
								part.Fatal(err)
							}

							if index >= 0 {
								arguments[slot] = run.storage.values[index]
							}
						}
					}
				case "CheckValues":
					for index, step := range f.session.plan.steps {
						if err = run.checkValue(f.session.plan.values[step.Outputs[0]], f.outputs[index], ErrInvalidModel); err != nil {
							part.Fatal(err)
						}
					}
				case "Publish":
					outputs, err = run.publish(f.session.plan.outputIndices)
				case "Clone":
					wrapped, err = run.cloneOutput(source)
				case "BackendClone":
					backend, err = source.Clone()
				case "PayloadCopy":
					copy(destination, values)
				case "RetainedLookup", "DirectLookup":
					for slot, value := range f.session.destinations.values {
						if value == nil {
							continue
						}

						backend = value
						if mode == "RetainedLookup" {
							backend = f.session.destinations.get(slot, Float32, shapes[slot])
						}
					}
				case "Acquisition":
					for index, step := range f.session.plan.steps {
						run.storage.slot = step.Index
						value := f.outputs[index]
						if backend, err = run.storage.destination(Float32, shapes[index], bytes[index], 0); err != nil {
							part.Fatal(err)
						}

						run.storage.values[step.Outputs[0]] = value
						run.storage.after(index)
					}
					run.storage.trim()
				case "Node09Dispatch":
					run.storage.slot = 9
					backend, err = f.session.plan.calls[9].run(run, f.args[9])
				case "ContextChecks":
					for range 337 {
						err = f.ctx.Err()
					}
				}
				if err != nil {
					part.Fatal(err)
				}
			}
			part.StopTimer()
			runtime.KeepAlive(arguments)
			runtime.KeepAlive(backend)
			runtime.KeepAlive(outputs)
			runtime.KeepAlive(wrapped)
			runtime.KeepAlive(created)
			runtime.KeepAlive(destination)
		})
	}
}

// pass19NilError is used only by the context-check ablation overlay.
func pass19NilError() (err error) { return nil }

func TestPass19Ledger(t *testing.T) {
	var (
		fixture       = newPass19Fixture(t)
		before, after runtime.MemStats
		run           *evaluation
		err           error
	)

	// Collect ordinary counters with no allocation-profile sampling or logging
	// inside each measured scope. Tiny identities can share allocator blocks.
	fixture.warm(t)
	const count = 1000
	bytes := make([]uint64, len(fixture.session.plan.steps)+3)
	objects := make([]uint64, len(bytes))
	for range count {
		runtime.ReadMemStats(&before)
		run = fixture.evaluation(t)
		runtime.ReadMemStats(&after)
		bytes[0] += after.TotalAlloc - before.TotalAlloc
		objects[0] += after.Mallocs - before.Mallocs
		if err != nil {
			t.Fatal(err)
		}

		runtime.ReadMemStats(&before)
		err = run.bind(fixture.inputs)
		runtime.ReadMemStats(&after)
		bytes[1] += after.TotalAlloc - before.TotalAlloc
		objects[1] += after.Mallocs - before.Mallocs
		if err != nil {
			t.Fatal(err)
		}

		for position, step := range fixture.session.plan.steps {
			runtime.ReadMemStats(&before)
			err = run.execute(step)
			runtime.ReadMemStats(&after)
			bytes[position+2] += after.TotalAlloc - before.TotalAlloc
			objects[position+2] += after.Mallocs - before.Mallocs
			if err != nil {
				t.Fatal(err)
			}

			run.storage.after(position)
		}
		run.storage.trim()
		runtime.ReadMemStats(&before)
		_, err = run.publish(fixture.session.plan.outputIndices)
		runtime.ReadMemStats(&after)
		bytes[len(bytes)-1] += after.TotalAlloc - before.TotalAlloc
		objects[len(bytes)-1] += after.Mallocs - before.Mallocs
		if err != nil {
			t.Fatal(err)
		}
	}
	for index := range bytes {
		t.Logf("scope=%d bytes=%.4f objects=%.4f", index, float64(bytes[index])/count, float64(objects[index])/count)
	}
}
