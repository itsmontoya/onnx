package onnx

import (
	"context"
	"encoding/json"
	"fmt"
	"runtime"
	"testing"

	nn13trace "github.com/itsmontoya/neuralnetwork/tools/nn13/trace"
	"github.com/itsmontoya/onnx/internal/decode"
)

func TestRT16PayloadMetrics(t *testing.T) {
	for _, fixture := range inferenceFixtures(t) {
		t.Run(fixture.Name, func(t *testing.T) {
			modelData, err := fixture.Model.read()
			if err != nil {
				t.Fatal(err)
			}

			inputs, inputErr := fixture.readInputs()
			if inputErr != nil {
				t.Fatal(inputErr)
			}

			ctx := context.Background()
			limits, limitErr := (Limits{}).normalized()
			if limitErr != nil {
				t.Fatal(limitErr)
			}

			nn13trace.Begin()
			wire, parseErr := decode.ParseModel(ctx, modelData, limits.wire())
			if parseErr != nil {
				t.Fatal(parseErr)
			}

			runtime.KeepAlive(wire)
			parse := nn13trace.End()
			nn13trace.Begin()
			model, loadErr := LoadBytes(ctx, modelData, Options{})
			if loadErr != nil {
				t.Fatal(loadErr)
			}

			if loadErr = model.Close(); loadErr != nil {
				t.Fatal(loadErr)
			}

			load := nn13trace.End()
			model, loadErr = LoadBytes(ctx, modelData, Options{})
			if loadErr != nil {
				t.Fatal(loadErr)
			}

			nn13trace.Begin(model, inputs)
			session, sessionErr := model.NewSession()
			if sessionErr != nil {
				t.Fatal(sessionErr)
			}

			outputs, runErr := session.Run(ctx, inputs, RunOptions{})
			if runErr != nil {
				t.Fatal(runErr)
			}

			if sessionErr = session.Close(); sessionErr != nil {
				t.Fatal(sessionErr)
			}

			first := nn13trace.End()
			if runErr = fixture.compareOutputs(outputs); runErr != nil {
				t.Fatal(runErr)
			}

			session, sessionErr = model.NewSession()
			if sessionErr != nil {
				t.Fatal(sessionErr)
			}

			retained, retainedErr := session.Run(ctx, inputs, RunOptions{})
			if retainedErr != nil {
				t.Fatal(retainedErr)
			}

			nn13trace.Begin(model, inputs, retained)
			outputs, runErr = session.Run(ctx, inputs, RunOptions{})
			if runErr != nil {
				t.Fatal(runErr)
			}

			warm := nn13trace.End()
			if sessionErr = session.Close(); sessionErr != nil {
				t.Fatal(sessionErr)
			}

			if loadErr = model.Close(); loadErr != nil {
				t.Fatal(loadErr)
			}

			if runErr = fixture.compareOutputs(outputs); runErr != nil {
				t.Fatal(runErr)
			}

			if retainedErr = fixture.compareOutputs(retained); retainedErr != nil {
				t.Fatal(retainedErr)
			}

			for phase, snapshot := range map[string]nn13trace.Snapshot{"Parse": parse, "Load": load, "First": first, "Warm": warm} {
				record := struct {
					Case     string             `json:"case"`
					Phase    string             `json:"phase"`
					Snapshot nn13trace.Snapshot `json:"metrics"`
				}{fixture.Name, phase, snapshot}
				encoded, encodeErr := json.Marshal(record)
				if encodeErr != nil {
					t.Fatal(encodeErr)
				}

				fmt.Printf("RT16METRIC %s\n", encoded)
			}
		})
	}
}
